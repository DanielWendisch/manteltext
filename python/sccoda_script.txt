from sccoda.util import cell_composition_data as dat

from sccoda.util import comp_ana as mod
import sccoda.datasets as scd

cell_counts =pd.read_csv(filepath_or_buffer= "..\\output\\meta_data_condition_per_sample_donor_day.csv")
data_grouping=pd.read_csv(filepath_or_buffer= "..\\output\\meta_data_condition_per_sample_donor_day_grouping.csv")

data_all = dat.from_pandas(cell_counts, covariate_columns=["sample"])
data_all.obs=data_grouping 

####################
data_severe=data_all[((data_all.obs.condition=="mild") | (data_all.obs.condition=="severe"))]
	
cell_types = data_severe.var.index
results_cycle = pd.DataFrame(index=cell_types, columns=["times_credible"]).fillna(0)
for ct in cell_types:
	print(f"Reference: {ct}")     
	model_temp = mod.CompositionalAnalysis(data_severe, formula="condition", reference_cell_type=ct)
	temp_results = model_temp.sample_hmc(num_results=20000)
        
	# Select credible effects
	cred_eff = temp_results.credible_effects()
	cred_eff.index = cred_eff.index.droplevel(level=0)
        
	# add up credible effects
	results_cycle["times_credible"] += cred_eff.astype("int")
	results_cycle.to_csv(ct+"_severe_vs_mild_times_credible.csv")
results_cycle.to_csv("severe_vs_control_times_credible.csv")
###############

for cond in ["severe","mild"]:
	data_severe=data_all[((data_all.obs.condition=="control") | (data_all.obs.condition==cond))]
	
	cell_types = data_severe.var.index
	results_cycle = pd.DataFrame(index=cell_types, columns=["times_credible"]).fillna(0)
	for ct in cell_types:
		print(f"Reference: {ct}")     
		model_temp = mod.CompositionalAnalysis(data_severe, formula="condition", reference_cell_type=ct)
		temp_results = model_temp.sample_hmc(num_results=20000)
        
		# Select credible effects
		cred_eff = temp_results.credible_effects()
		cred_eff.index = cred_eff.index.droplevel(level=0)
        
		# add up credible effects
		results_cycle["times_credible"] += cred_eff.astype("int")
		results_cycle.to_csv(ct+cond+"_vs_control_times_credible.csv")
	results_cycle.to_csv(cond+"_vs_control_times_credible.csv")
